/**
 * @param {Number} value
 * @param {Number} min
 * @param {Number} max
 *
 * @description
 * Clamps a value between a minimum and maximum value
 */
function clampValue(value, min, max) {
  if (value < min) {
    return min;
  }
  if (value > max) {
    return max;
  }
  return value;
}
